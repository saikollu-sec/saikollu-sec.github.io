Detection Rule Showcase
What is this? — A real KQL detection rule written for Microsoft Sentinel. KQL is the query language used to build detection logic in Sentinel. This rule detects impossible travel combined with MFA bypass — a common attack pattern where stolen credentials are used simultaneously from two geographically distant locations. Mapped to MITRE ATT&CK T1078 (Valid Accounts) and T1110 (Brute Force).
impossible_travel_mfa_bypass.kql · Microsoft Sentinel · Author: Sai Kollu
KQLT1078 · T1110HIGH
// Detects impossible travel + MFA bypass — common credential theft pattern
// MITRE: T1078 (Valid Accounts) · T1110 (Brute Force) · Author: Sai Kollu

let timeframe = 1h;
let distance_threshold_km = 500;

SigninLogs
| where TimeGenerated > ago(timeframe)
| where ResultType == "0"                    // successful sign-ins only
| where AuthenticationRequirement == "singleFactorAuthentication"
| project TimeGenerated, UserPrincipalName,
          IPAddress, Latitude, Longitude, AppDisplayName
| join kind=inner (
    SigninLogs
    | where TimeGenerated > ago(timeframe)
    | project UserPrincipalName,
              IPAddress2 = IPAddress,
              Lat2 = Latitude, Lon2 = Longitude,
              Time2 = TimeGenerated
) on UserPrincipalName
| where IPAddress != IPAddress2
    and abs(TimeGenerated - Time2) < 30m
| extend distance_km = geo_distance_2points(
    Longitude, Latitude, Lon2, Lat2) / 1000
| where distance_km > distance_threshold_km
| extend
    AlertSeverity  = "High",
    MITRETactic    = "Initial Access · Credential Access",
    Recommendation = "Revoke session · Enforce MFA · Block source IP"
| project TimeGenerated, UserPrincipalName,
          IPAddress, IPAddress2, distance_km,
          AlertSeverity, MITRETactic, Recommendation
| order by distance_km desc
